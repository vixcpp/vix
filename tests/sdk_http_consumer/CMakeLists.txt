cmake_minimum_required(VERSION 3.23)
project(vix_http_installed_consumer LANGUAGES CXX)
find_package(Vix CONFIG REQUIRED COMPONENTS HTTPExperimental)
add_executable(vix_http_installed_consumer main.cpp)
target_compile_features(vix_http_installed_consumer PRIVATE cxx_std_20)
target_link_libraries(vix_http_installed_consumer PRIVATE vix::http_experimental)
add_executable(vix_http_installed_client_example ../../domains/http/examples/client.cpp)
target_compile_features(vix_http_installed_client_example PRIVATE cxx_std_20)
target_link_libraries(vix_http_installed_client_example PRIVATE vix::http_experimental)
add_executable(vix_http_installed_server_example ../../domains/http/examples/server.cpp)
target_compile_features(vix_http_installed_server_example PRIVATE cxx_std_20)
target_link_libraries(vix_http_installed_server_example PRIVATE vix::http_experimental)
