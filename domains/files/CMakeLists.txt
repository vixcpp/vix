cmake_minimum_required(VERSION 3.23)
include(GNUInstallDirs)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vix_files LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  option(VIX_BUILD_TESTS "Build focused Vix Files tests" OFF)
  add_subdirectory(../../standard standard_build)
endif()

if (WIN32)
  set(VIX_FILES_PROVIDER src/FilesWindows.cpp)
elseif (ANDROID)
  if (NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
    message(FATAL_ERROR "vix::files Android provider is configured for arm64-v8a and x86_64; got '${ANDROID_ABI}'.")
  endif()
  set(_vix_files_android_api "")
  if (DEFINED ANDROID_PLATFORM_LEVEL)
    set(_vix_files_android_api "${ANDROID_PLATFORM_LEVEL}")
  elseif (ANDROID_PLATFORM MATCHES "android-([0-9]+)$")
    set(_vix_files_android_api "${CMAKE_MATCH_1}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "^([0-9]+)$")
    set(_vix_files_android_api "${CMAKE_MATCH_1}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "android-([0-9]+)$")
    set(_vix_files_android_api "${CMAKE_MATCH_1}")
  endif()
  if (_vix_files_android_api STREQUAL "")
    message(FATAL_ERROR "Cannot determine the Android API level from the active NDK toolchain.")
  endif()
  if (_vix_files_android_api LESS 21)
    message(FATAL_ERROR "vix::files Android provider requires Android API 21 or newer; got API ${_vix_files_android_api}.")
  endif()
  # Android NDK/Bionic exposes the Linux-family file and directory APIs used
  # by this provider; host access remains governed by Android permissions.
  set(VIX_FILES_PROVIDER src/FilesPosix.cpp)
elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(VIX_FILES_PROVIDER src/FilesPosix.cpp)
elseif (CMAKE_SYSTEM_NAME STREQUAL "WASI")
  if (NOT CMAKE_CXX_COMPILER_TARGET STREQUAL "wasm32-wasip2")
    message(FATAL_ERROR "vix::files WASI provider requires the wasm32-wasip2 target.")
  endif()
  # wasi-libc maps these calls to the Preview 2 filesystem capability model.
  set(VIX_FILES_PROVIDER src/FilesPosix.cpp)
else()
  message(FATAL_ERROR "vix::files has no provider for ${CMAKE_SYSTEM_NAME}")
endif()

add_library(vix_files src/FilesCommon.cpp ${VIX_FILES_PROVIDER})
add_library(vix::files ALIAS vix_files)

target_compile_features(vix_files PUBLIC cxx_std_20)
target_include_directories(vix_files PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(vix_files PUBLIC vix::standard)

install(TARGETS vix_files EXPORT VixTargets)
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  FILES_MATCHING PATTERN "*.hpp" PATTERN "detail/*" EXCLUDE)

if (VIX_BUILD_TESTS)
  if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
  endif()
  add_subdirectory(tests)
endif()
