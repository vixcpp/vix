cmake_minimum_required(VERSION 3.23)
include(GNUInstallDirs)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vix_process LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  option(VIX_BUILD_TESTS "Build focused Vix Process tests" OFF)
  add_subdirectory(../../standard standard_build)
  add_subdirectory(../files files_build)
endif()

if (WIN32)
  set(VIX_PROCESS_PROVIDER src/ProcessWindows.cpp)
elseif (CMAKE_SYSTEM_NAME STREQUAL "WASI")
  if (NOT CMAKE_CXX_COMPILER_TARGET STREQUAL "wasm32-wasip2")
    message(FATAL_ERROR "vix::process WASI provider requires wasm32-wasip2.")
  endif()
  set(VIX_PROCESS_PROVIDER src/ProcessWasiUnsupported.cpp)
elseif (ANDROID)
  if (NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
    message(FATAL_ERROR "vix::process Android provider supports arm64-v8a and x86_64; got '${ANDROID_ABI}'.")
  endif()
  set(_vix_process_android_api "")
  if (DEFINED ANDROID_PLATFORM_LEVEL)
    set(_vix_process_android_api "${ANDROID_PLATFORM_LEVEL}")
  elseif (ANDROID_PLATFORM MATCHES "android-([0-9]+)$")
    set(_vix_process_android_api "${CMAKE_MATCH_1}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "^([0-9]+)$")
    set(_vix_process_android_api "${CMAKE_SYSTEM_VERSION}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "android-([0-9]+)$")
    set(_vix_process_android_api "${CMAKE_MATCH_1}")
  endif()
  if (_vix_process_android_api STREQUAL "" OR _vix_process_android_api LESS 21)
    message(FATAL_ERROR "vix::process Android provider requires a determinable API level of 21 or newer.")
  endif()
  set(VIX_PROCESS_PROVIDER src/ProcessPosix.cpp)
  find_package(Threads REQUIRED)
elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(VIX_PROCESS_PROVIDER src/ProcessPosix.cpp)
  find_package(Threads REQUIRED)
else()
  message(FATAL_ERROR "vix::process has no provider for ${CMAKE_SYSTEM_NAME}.")
endif()

add_library(vix_process ${VIX_PROCESS_PROVIDER})
add_library(vix::process ALIAS vix_process)
target_compile_features(vix_process PUBLIC cxx_std_20)
target_include_directories(vix_process PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(vix_process PUBLIC vix::standard vix::files)
if (APPLE)
  # Expose Darwin's pipe2 declaration for the atomic CLOEXEC path on macOS 10.15+.
  target_compile_definitions(vix_process PRIVATE _DARWIN_C_SOURCE)
endif()
if (NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  target_link_libraries(vix_process PRIVATE Threads::Threads)
endif()

install(TARGETS vix_process EXPORT VixTargets)
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  FILES_MATCHING PATTERN "*.hpp")

if (VIX_BUILD_TESTS)
  if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
  endif()
  add_subdirectory(tests)
endif()
