cmake_minimum_required(VERSION 3.23)
include(GNUInstallDirs)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vix_concurrency LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  option(VIX_BUILD_TESTS "Build focused Vix Concurrency tests" OFF)
  add_subdirectory(../../standard standard_build)
endif()

if (CMAKE_SYSTEM_NAME STREQUAL "WASI")
  if (NOT CMAKE_CXX_COMPILER_TARGET STREQUAL "wasm32-wasip2")
    message(FATAL_ERROR "vix::concurrency WASI provider requires wasm32-wasip2.")
  endif()
  set(VIX_CONCURRENCY_PROVIDER src/ExecutionResourceWasi.cpp)
elseif (ANDROID)
  if (NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
    message(FATAL_ERROR "vix::concurrency Android provider supports arm64-v8a and x86_64; got '${ANDROID_ABI}'.")
  endif()
  set(_vix_concurrency_android_api "")
  if (DEFINED ANDROID_PLATFORM_LEVEL)
    set(_vix_concurrency_android_api "${ANDROID_PLATFORM_LEVEL}")
  elseif (ANDROID_PLATFORM MATCHES "android-([0-9]+)$")
    set(_vix_concurrency_android_api "${CMAKE_MATCH_1}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "^([0-9]+)$")
    set(_vix_concurrency_android_api "${CMAKE_SYSTEM_VERSION}")
  elseif (CMAKE_SYSTEM_VERSION MATCHES "android-([0-9]+)$")
    set(_vix_concurrency_android_api "${CMAKE_MATCH_1}")
  endif()
  if (_vix_concurrency_android_api STREQUAL "" OR _vix_concurrency_android_api LESS 21)
    message(FATAL_ERROR "vix::concurrency Android provider requires a determinable API level of 21 or newer.")
  endif()
  set(VIX_CONCURRENCY_PROVIDER src/ExecutionResource.cpp)
  find_package(Threads REQUIRED)
elseif (CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|Windows)$")
  set(VIX_CONCURRENCY_PROVIDER src/ExecutionResource.cpp)
  find_package(Threads REQUIRED)
else()
  message(FATAL_ERROR "vix::concurrency has no provider for ${CMAKE_SYSTEM_NAME}.")
endif()

add_library(vix_concurrency ${VIX_CONCURRENCY_PROVIDER})
add_library(vix::concurrency ALIAS vix_concurrency)
target_compile_features(vix_concurrency PUBLIC cxx_std_20)
target_include_directories(vix_concurrency PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(vix_concurrency PUBLIC vix::standard)
if (NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  target_link_libraries(vix_concurrency PRIVATE Threads::Threads)
endif()

install(TARGETS vix_concurrency EXPORT VixTargets)
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  FILES_MATCHING PATTERN "*.hpp")

if (VIX_BUILD_TESTS)
  if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
  endif()
  add_subdirectory(tests)
endif()
